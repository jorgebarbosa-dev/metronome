## Current Accessibility State

**Existing good practices:**
- All interactive elements use native `<button>` tags
- `aria-label` present on all controls
- `aria-pressed` on toggle buttons (PlayButton, AccentToggle, SoundSelector, SubdivisionSelector)
- `role="group"` on grouped controls (SoundSelector, SubdivisionSelector)
- `role="status"` on dynamic displays (BpmDisplay, BeatIndicators, BeatNumberDisplay)
- `aria-live="polite"` on BeatNumberDisplay
- `aria-hidden="true"` on decorative icons
- `focus:outline-none focus:ring-2/4` classes on interactive elements
- Dark mode support via `dark:` Tailwind classes
- `min-h-[44px]` touch targets

**Gaps to address:**
1. No global keyboard shortcut for play/pause (Space key)
2. No `aria-live` on BPM display — screen reader doesn't announce BPM changes
3. No `aria-live` for play state changes — screen reader doesn't announce start/stop
4. No `aria-live` for time signature changes
5. No arrow key navigation within SoundSelector and SubdivisionSelector button groups
6. No `focus-visible` distinction — focus rings may show on mouse clicks
7. Color contrast not explicitly verified against WCAG AA

```typescript
import { useEffect, useCallback } from 'react';
import { useMetronome } from '../context/MetronomeContext';

export function useKeyboardShortcuts() {
  const { state, dispatch } = useMetronome();

  const handleKeyDown = useCallback((e: KeyboardEvent) => {
    // Space key for play/pause, but only when not typing in an input
    if (e.key === ' ' && e.target instanceof HTMLElement) {
      const tagName = e.target.tagName.toLowerCase();
      const isInput = tagName === 'input' || tagName === 'textarea' || tagName === 'select';
      const isContentEditable = e.target.isContentEditable;
      
      if (!isInput && !isContentEditable) {
        e.preventDefault();
        dispatch({ type: 'TOGGLE_PLAY' });
      }
    }
  }, [dispatch]);

  useEffect(() => {
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [handleKeyDown]);
}
```

```typescript
import { useRef } from 'react'; // add to imports

export function SoundSelector() {
  const { state, dispatch } = useMetronome();
  const { selectedSound } = state;
  const buttonRefs = useRef<(HTMLButtonElement | null)[]>([]);

  const sounds = [...]; // existing array

  const handleKeyDown = (e: React.KeyboardEvent<HTMLButtonElement>, index: number) => {
    if (e.key === 'ArrowRight') {
      e.preventDefault();
      const nextIndex = (index + 1) % sounds.length;
      buttonRefs.current[nextIndex]?.focus();
    } else if (e.key === 'ArrowLeft') {
      e.preventDefault();
      const prevIndex = (index - 1 + sounds.length) % sounds.length;
      buttonRefs.current[prevIndex]?.focus();
    } else if (e.key === 'Home') {
      e.preventDefault();
      buttonRefs.current[0]?.focus();
    } else if (e.key === 'End') {
      e.preventDefault();
      buttonRefs.current[sounds.length - 1]?.focus();
    }
  };

  return (
    <div className="flex flex-col items-center gap-2" role="group" aria-label="Sound selection">
      <span className="text-sm font-medium text-gray-600 dark:text-gray-400">Sound</span>
      <div className="flex gap-1 p-1 bg-gray-100 dark:bg-gray-800 rounded-lg">
        {sounds.map((sound, index) => {
          const isSelected = selectedSound === sound.name;
          return (
            <button
              key={sound.name}
              ref={(el) => { buttonRefs.current[index] = el; }}
              onClick={() => dispatch({ type: 'SET_SOUND', payload: sound.name as SoundName })}
              onKeyDown={(e) => handleKeyDown(e, index)}
              // ... existing className and aria attributes
            >
              {sound.label}
            </button>
          );
        })}
      </div>
    </div>
  );
}
```

   - Also add `tabIndex={isSelected ? 0 : -1}` to implement roving tabindex pattern: only the selected button is in the natural tab order, arrow keys move focus within the group. This is the standard WAI-ARIA pattern for radio button groups.
